cmake_minimum_required(VERSION 3.22)
if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS version")
endif()
project(hearth_desktop_runtime LANGUAGES C CXX)
if(APPLE AND HEARTH_DESKTOP_METAL)
 enable_language(OBJC OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN YES)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
 add_compile_options(/utf-8)
endif()
find_package(JNI REQUIRED)
set(CPP "${HEARTH_ROOT}/common-jni/src/main/cpp")
set(DESKTOP_CPP "${HEARTH_ROOT}/desktop-runtime/src/main/cpp")
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ${HEARTH_DESKTOP_METAL} CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(HEARTH_STAGE STREQUAL "whisper")
 set(WHISPER "${CPP}/whisper/upstream")
 add_subdirectory("${WHISPER}/ggml" ggml)
 add_library(whisper_core STATIC "${WHISPER}/src/whisper.cpp")
 target_compile_definitions(whisper_core PRIVATE WHISPER_VERSION="1.8.2")
 target_include_directories(whisper_core PUBLIC "${WHISPER}/include")
 target_link_libraries(whisper_core PRIVATE ggml)
 add_library(hearth_desktop_whisper SHARED "${DESKTOP_CPP}/whisper_jni.cpp")
 target_compile_definitions(hearth_desktop_whisper PRIVATE HEARTH_DESKTOP_METAL=$<BOOL:${HEARTH_DESKTOP_METAL}>)
 target_include_directories(hearth_desktop_whisper PRIVATE ${JNI_INCLUDE_DIRS} "${CPP}/common" "${WHISPER}/include")
 target_link_libraries(hearth_desktop_whisper PRIVATE whisper_core ggml)
 set(OUTPUT_TARGET hearth_desktop_whisper)
elseif(HEARTH_STAGE STREQUAL "translation")
 set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
 set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
 set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
 add_subdirectory("${HEARTH_LLAMA_ROOT}" llama)
 add_library(transiber_translation SHARED "${HEARTH_TRANSLATION_JNI_SOURCE}")
 target_compile_definitions(transiber_translation PRIVATE HEARTH_DESKTOP_METAL=$<BOOL:${HEARTH_DESKTOP_METAL}>)
 target_include_directories(transiber_translation PRIVATE ${JNI_INCLUDE_DIRS} "${DESKTOP_CPP}" "${CPP}/translation" "${CPP}/common")
 target_link_libraries(transiber_translation PRIVATE llama)
 set(OUTPUT_TARGET transiber_translation)
elseif(HEARTH_STAGE STREQUAL "supertonic")
 if(NOT EXISTS "${HEARTH_ORT_ROOT}/VERSION_NUMBER")
  message(FATAL_ERROR "Set HEARTH_ORT_ROOT to the extracted official ONNX Runtime 1.20.1 host SDK")
 endif()
 file(READ "${HEARTH_ORT_ROOT}/VERSION_NUMBER" ORT_VERSION)
 string(STRIP "${ORT_VERSION}" ORT_VERSION)
 if(NOT ORT_VERSION STREQUAL "1.20.1")
  message(FATAL_ERROR "The shared Supertonic source pins ONNX Runtime 1.20.1")
 endif()
 add_library(onnx_host SHARED IMPORTED)
 if(WIN32)
  set_target_properties(onnx_host PROPERTIES IMPORTED_LOCATION "${HEARTH_ORT_ROOT}/lib/onnxruntime.dll" IMPORTED_IMPLIB "${HEARTH_ORT_ROOT}/lib/onnxruntime.lib")
 elseif(APPLE)
  set_target_properties(onnx_host PROPERTIES IMPORTED_LOCATION "${HEARTH_ORT_ROOT}/lib/libonnxruntime.1.20.1.dylib")
 else()
  set_target_properties(onnx_host PROPERTIES IMPORTED_LOCATION "${HEARTH_ORT_ROOT}/lib/libonnxruntime.so")
 endif()
 set(VOICE "${HEARTH_ROOT}/hearth-tts-supertonic/src/main/cpp/voice")
 add_library(hearth_supertonic_jni SHARED "${HEARTH_SUPERTONIC_JNI_SOURCE}" "${HEARTH_SUPERTONIC_SOURCE}")
 target_include_directories(hearth_supertonic_jni PRIVATE ${JNI_INCLUDE_DIRS} "${VOICE}" "${DESKTOP_CPP}" "${CPP}/common" "${CPP}/onnx/include")
 target_link_libraries(hearth_supertonic_jni PRIVATE onnx_host)
 set_target_properties(hearth_supertonic_jni PROPERTIES BUILD_RPATH "${HEARTH_ORT_ROOT}/lib" INSTALL_RPATH "@loader_path")
 set(OUTPUT_TARGET hearth_supertonic_jni)
elseif(HEARTH_STAGE STREQUAL "onnx")
 if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
  message(FATAL_ERROR "The Qwen3-ASR desktop runtime is built only for macOS arm64")
 endif()
 if(NOT EXISTS "${HEARTH_ORT_ROOT}/VERSION_NUMBER")
  message(FATAL_ERROR "Set HEARTH_ORT_ROOT to the extracted official ONNX Runtime 1.20.1 host SDK")
 endif()
 file(READ "${HEARTH_ORT_ROOT}/VERSION_NUMBER" ORT_VERSION)
 string(STRIP "${ORT_VERSION}" ORT_VERSION)
 if(NOT ORT_VERSION STREQUAL "1.20.1")
  message(FATAL_ERROR "The desktop bundle ships ONNX Runtime 1.20.1")
 endif()
 if(NOT EXISTS "${HEARTH_SHERPA_ROOT}/include/sherpa-onnx/c-api/c-api.h" OR
    NOT EXISTS "${HEARTH_SHERPA_ROOT}/lib/libsherpa-onnx.a")
  message(FATAL_ERROR "Set HEARTH_SHERPA_ROOT to the pinned static sherpa-onnx install tree")
 endif()
 add_library(onnx_host SHARED IMPORTED)
 if(WIN32)
  set_target_properties(onnx_host PROPERTIES IMPORTED_LOCATION "${HEARTH_ORT_ROOT}/lib/onnxruntime.dll" IMPORTED_IMPLIB "${HEARTH_ORT_ROOT}/lib/onnxruntime.lib")
 elseif(APPLE)
  set_target_properties(onnx_host PROPERTIES IMPORTED_LOCATION "${HEARTH_ORT_ROOT}/lib/libonnxruntime.1.20.1.dylib")
 else()
  set_target_properties(onnx_host PROPERTIES IMPORTED_LOCATION "${HEARTH_ORT_ROOT}/lib/libonnxruntime.so")
 endif()
 # The Android backend and the shared ONNX framework are compiled unchanged.
 # sherpa-onnx is statically linked; ONNX Runtime stays the shipped 1.20.1 dylib.
 add_library(hearth_desktop_qwen3_asr SHARED
   "${DESKTOP_CPP}/qwen_jni.cpp"
   "${CPP}/speech/backends/qwen_backend.cpp"
   "${CPP}/onnx/onnx_engine.cpp"
   "${CPP}/common/logging.cpp"
   "${CPP}/common/json_utils.cpp"
   "${CPP}/common/engine_interface.cpp"
   "${CPP}/common/audio_utils.cpp"
 )
 target_compile_definitions(hearth_desktop_qwen3_asr PRIVATE ONNX_AVAILABLE=1)
 target_include_directories(hearth_desktop_qwen3_asr PRIVATE
   ${JNI_INCLUDE_DIRS} "${DESKTOP_CPP}" "${CPP}/common" "${CPP}/speech" "${CPP}/speech/backends"
   "${CPP}/onnx" "${CPP}/onnx/include" "${CPP}/onnx/include/onnxruntime"
   "${HEARTH_SHERPA_ROOT}/include")
 target_link_libraries(hearth_desktop_qwen3_asr PRIVATE onnx_host "${HEARTH_SHERPA_ROOT}/lib/libsherpa-onnx.a" "-framework Foundation")
 set_target_properties(hearth_desktop_qwen3_asr PROPERTIES BUILD_RPATH "${HEARTH_ORT_ROOT}/lib" INSTALL_RPATH "@loader_path")
 set(OUTPUT_TARGET hearth_desktop_qwen3_asr)
elseif(HEARTH_STAGE STREQUAL "moss")
 if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
  message(FATAL_ERROR "The MOSS desktop voice runtime is built only for macOS arm64")
 endif()
 if(NOT EXISTS "${HEARTH_ORT_ROOT}/VERSION_NUMBER")
  message(FATAL_ERROR "Set HEARTH_ORT_ROOT to the extracted official ONNX Runtime 1.20.1 host SDK")
 endif()
 file(READ "${HEARTH_ORT_ROOT}/VERSION_NUMBER" ORT_VERSION)
 string(STRIP "${ORT_VERSION}" ORT_VERSION)
 if(NOT ORT_VERSION STREQUAL "1.20.1")
  message(FATAL_ERROR "The desktop bundle ships ONNX Runtime 1.20.1")
 endif()
 if(NOT DEFINED HEARTH_MOSS_CPP)
  set(HEARTH_MOSS_CPP "${HEARTH_ROOT}/hearth-tts-moss/src/main/cpp")
 endif()
 if(NOT EXISTS "${HEARTH_SENTENCEPIECE_SOURCE}/src/sentencepiece_processor.h")
  message(FATAL_ERROR "Set HEARTH_SENTENCEPIECE_SOURCE to the pinned SentencePiece 0.2.0 source")
 endif()
 add_library(onnx_host SHARED IMPORTED)
 set_target_properties(onnx_host PROPERTIES IMPORTED_LOCATION "${HEARTH_ORT_ROOT}/lib/libonnxruntime.1.20.1.dylib")
 set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
 set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
 set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
 set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
 set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
 add_subdirectory("${HEARTH_SENTENCEPIECE_SOURCE}" spm EXCLUDE_FROM_ALL)
 add_library(hearth_desktop_moss SHARED "${DESKTOP_CPP}/moss_jni.cpp" "${HEARTH_MOSS_CPP}/moss.cpp")
 target_include_directories(hearth_desktop_moss PRIVATE ${JNI_INCLUDE_DIRS} "${HEARTH_MOSS_CPP}" "${HEARTH_MOSS_CPP}/vendor" "${HEARTH_SENTENCEPIECE_SOURCE}/src" "${CPP}/common" "${CPP}/onnx/include")
 target_link_libraries(hearth_desktop_moss PRIVATE sentencepiece-static onnx_host)
 set_target_properties(hearth_desktop_moss PROPERTIES BUILD_RPATH "${HEARTH_ORT_ROOT}/lib" INSTALL_RPATH "@loader_path")
 set(OUTPUT_TARGET hearth_desktop_moss)
elseif(HEARTH_STAGE STREQUAL "omnivoice")
 if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
  message(FATAL_ERROR "The OmniVoice desktop voice runtime is built only for macOS arm64")
 endif()
 if(NOT DEFINED HEARTH_OMNIVOICE_SOURCE OR NOT EXISTS "${HEARTH_OMNIVOICE_SOURCE}/src/omnivoice.cpp" OR
    NOT EXISTS "${HEARTH_OMNIVOICE_SOURCE}/ggml/CMakeLists.txt")
  message(FATAL_ERROR "Set HEARTH_OMNIVOICE_SOURCE to the pinned omnivoice.cpp checkout")
 endif()
 # OmniVoice statically owns one pinned ggml build of its own. It is never linked
 # into whisper/parakeet/NeMo dylibs and does not use Metal GPU offload in this stage.
 set(GGML_METAL OFF CACHE BOOL "" FORCE)
 set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
 set(GGML_BLAS OFF CACHE BOOL "" FORCE)
 set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
 set(GGML_CUDA OFF CACHE BOOL "" FORCE)
 set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
 set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 add_compile_definitions(GGML_MAX_NAME=128)
 # Pin the embedded source identity for the runtime and its static ggml so two clean
 # checkouts at the same revisions produce the same dylib, independent of checkout path.
 add_compile_options(
   "-ffile-prefix-map=${HEARTH_OMNIVOICE_SOURCE}=omnivoice"
   "-fdebug-prefix-map=${HEARTH_OMNIVOICE_SOURCE}=omnivoice")
 add_subdirectory("${HEARTH_OMNIVOICE_SOURCE}/ggml" ggml-omnivoice)
 find_package(Threads REQUIRED)
 execute_process(COMMAND "${CMAKE_COMMAND}" "-DSRC_DIR=${HEARTH_OMNIVOICE_SOURCE}" "-DOUTPUT=${CMAKE_BINARY_DIR}/version.h" -P "${HEARTH_OMNIVOICE_SOURCE}/tools/version.cmake" COMMAND_ERROR_IS_FATAL ANY)
 add_library(hearth_desktop_omnivoice SHARED
   "${DESKTOP_CPP}/omnivoice_jni.cpp"
   "${HEARTH_OMNIVOICE_SOURCE}/src/omnivoice.cpp"
   "${HEARTH_OMNIVOICE_SOURCE}/src/pipeline-codec.cpp"
   "${HEARTH_OMNIVOICE_SOURCE}/src/pipeline-tts.cpp"
 )
 target_compile_definitions(hearth_desktop_omnivoice PRIVATE OMNIVOICE_BUILD=1)
 target_include_directories(hearth_desktop_omnivoice PRIVATE
   ${JNI_INCLUDE_DIRS} "${DESKTOP_CPP}" "${CPP}/common"
   "${HEARTH_OMNIVOICE_SOURCE}/src" "${HEARTH_OMNIVOICE_SOURCE}/ggml/include"
   "${HEARTH_ROOT}/hearth-tts-omnivoice/src/main/cpp/vendor" "${CMAKE_BINARY_DIR}")
 target_link_libraries(hearth_desktop_omnivoice PRIVATE ggml Threads::Threads)
 set(OUTPUT_TARGET hearth_desktop_omnivoice)
elseif(HEARTH_STAGE STREQUAL "chatterbox")
 if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
  message(FATAL_ERROR "The Chatterbox desktop voice runtime is built only for macOS arm64")
 endif()
 if(NOT EXISTS "${HEARTH_ORT_ROOT}/VERSION_NUMBER")
  message(FATAL_ERROR "Set HEARTH_ORT_ROOT to the extracted official ONNX Runtime 1.20.1 host SDK")
 endif()
 file(READ "${HEARTH_ORT_ROOT}/VERSION_NUMBER" ORT_VERSION)
 string(STRIP "${ORT_VERSION}" ORT_VERSION)
 if(NOT ORT_VERSION STREQUAL "1.20.1")
  message(FATAL_ERROR "The desktop bundle ships ONNX Runtime 1.20.1")
 endif()
 if(NOT DEFINED HEARTH_CHATTERBOX_SOURCE OR NOT EXISTS "${HEARTH_CHATTERBOX_SOURCE}/src/audio.cpp" OR
    NOT EXISTS "${HEARTH_CHATTERBOX_SOURCE}/include/chatterbox/watermark.hpp")
  message(FATAL_ERROR "Set HEARTH_CHATTERBOX_SOURCE to the pinned chatterboxcpp checkout")
 endif()
 if(NOT DEFINED HEARTH_CHATTERBOX_WATERMARK OR NOT EXISTS "${HEARTH_CHATTERBOX_WATERMARK}")
  message(FATAL_ERROR "Set HEARTH_CHATTERBOX_WATERMARK to the cancellation-patched Perth source")
 endif()
 if(NOT DEFINED HEARTH_BPE_LIBRARY OR NOT EXISTS "${HEARTH_BPE_LIBRARY}")
  message(FATAL_ERROR "Set HEARTH_BPE_LIBRARY to the pinned Rust byte-BPE static library")
 endif()
 if(NOT DEFINED HEARTH_CHATTERBOX_CPP)
  set(HEARTH_CHATTERBOX_CPP "${HEARTH_ROOT}/hearth-tts-chatterbox/src/main/cpp")
 endif()
 add_library(onnx_host SHARED IMPORTED)
 set_target_properties(onnx_host PROPERTIES IMPORTED_LOCATION "${HEARTH_ORT_ROOT}/lib/libonnxruntime.1.20.1.dylib")
 add_library(hearth_bpe STATIC IMPORTED)
 set_target_properties(hearth_bpe PROPERTIES IMPORTED_LOCATION "${HEARTH_BPE_LIBRARY}")
 # The runtime statically owns its Rust byte-BPE and links only the shared ORT dylib.
 add_library(hearth_desktop_chatterbox SHARED
   "${DESKTOP_CPP}/chatterbox_jni.cpp"
   "${HEARTH_CHATTERBOX_CPP}/chatterbox.cpp"
   "${HEARTH_CHATTERBOX_SOURCE}/src/audio.cpp"
   "${HEARTH_CHATTERBOX_WATERMARK}"
 )
 target_compile_definitions(hearth_desktop_chatterbox PRIVATE HEARTH_CHATTERBOX_TRACE=1)
 target_include_directories(hearth_desktop_chatterbox PRIVATE
   ${JNI_INCLUDE_DIRS} "${HEARTH_CHATTERBOX_CPP}" "${HEARTH_CHATTERBOX_CPP}/vendor"
   "${HEARTH_CHATTERBOX_SOURCE}/include" "${CPP}/common" "${CPP}/onnx/include")
 target_link_libraries(hearth_desktop_chatterbox PRIVATE onnx_host hearth_bpe)
 set_target_properties(hearth_desktop_chatterbox PROPERTIES BUILD_RPATH "${HEARTH_ORT_ROOT}/lib" INSTALL_RPATH "@loader_path")
 set(OUTPUT_TARGET hearth_desktop_chatterbox)
elseif(HEARTH_STAGE STREQUAL "backend")
 if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
  message(FATAL_ERROR "The desktop speech backend bridge is built only for macOS arm64")
 endif()
 # The bridge dlopens a pinned backend dylib by absolute path; no SDK is linked here.
 add_library(hearth_desktop_backend_speech SHARED "${DESKTOP_CPP}/backend_speech_jni.cpp")
 target_include_directories(hearth_desktop_backend_speech PRIVATE ${JNI_INCLUDE_DIRS} "${DESKTOP_CPP}" "${CPP}/common" "${CPP}/speech")
 target_link_libraries(hearth_desktop_backend_speech PRIVATE ${CMAKE_DL_LIBS})
 set(OUTPUT_TARGET hearth_desktop_backend_speech)
elseif(HEARTH_STAGE STREQUAL "vosk")
 if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
  message(FATAL_ERROR "The Vosk desktop backend is built only for macOS arm64")
 endif()
 # The pinned official libvosk.dylib is never linked here: this adapter dlopen()s
 # its sibling by absolute path at create() so the runtime stays a verified bundle
 # file and the adapter carries no Vosk load command.
 add_library(hearth_vosk SHARED
   "${CPP}/speech/backends/vosk_backend.cpp"
   "${CPP}/vosk/vosk_engine.cpp"
   "${CPP}/common/logging.cpp"
   "${CPP}/common/json_utils.cpp"
   "${CPP}/common/engine_interface.cpp"
   "${CPP}/common/audio_utils.cpp"
 )
 target_compile_definitions(hearth_vosk PRIVATE VOSK_AVAILABLE=1)
 target_include_directories(hearth_vosk PRIVATE
   "${CPP}" "${CPP}/common" "${CPP}/speech" "${CPP}/speech/backends" "${CPP}/vosk")
 target_link_libraries(hearth_vosk PRIVATE ${CMAKE_DL_LIBS})
 set(OUTPUT_TARGET hearth_vosk)
else()
 message(FATAL_ERROR "HEARTH_STAGE must be whisper, translation, supertonic, onnx, moss, omnivoice, chatterbox, backend or vosk")
endif()
if(APPLE AND HEARTH_DESKTOP_METAL AND NOT HEARTH_STAGE STREQUAL "supertonic" AND NOT HEARTH_STAGE STREQUAL "onnx" AND NOT HEARTH_STAGE STREQUAL "moss" AND NOT HEARTH_STAGE STREQUAL "omnivoice" AND NOT HEARTH_STAGE STREQUAL "chatterbox" AND NOT HEARTH_STAGE STREQUAL "backend" AND NOT HEARTH_STAGE STREQUAL "vosk")
 target_sources(${OUTPUT_TARGET} PRIVATE "${DESKTOP_CPP}/metal_device.mm")
endif()
if(APPLE)
 target_link_options(${OUTPUT_TARGET} PRIVATE "-Wl,-exported_symbols_list,${CMAKE_CURRENT_LIST_DIR}/${HEARTH_STAGE}.exports")
elseif(UNIX)
 target_link_options(${OUTPUT_TARGET} PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
endif()
install(TARGETS ${OUTPUT_TARGET} RUNTIME DESTINATION . COMPONENT hearth-desktop LIBRARY DESTINATION . COMPONENT hearth-desktop)
