# Windows x64 twin of scripts/speech/transcribepsy.cmake (CPU only). That recipe is part of
# the Android runtime receipt, so the MSVC differences live here instead of in it.
cmake_minimum_required(VERSION 3.26)
project(hearth_transcribepsy_windows LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option BACKEND_DL CPU_ALL_VARIANTS NATIVE OPENMP METAL BLAS ACCELERATE CUDA VULKAN OPENCL CPU_KLEIDIAI)
  set(GGML_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(PARAKEET_OPENMP OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_EXECUTABLES OFF CACHE BOOL "" FORCE)
set(PARAKEET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PARAKEET_INSTALL OFF CACHE BOOL "" FORCE)
set(PARAKEET_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
add_compile_options(/utf-8)
add_subdirectory("${TRANSCRIBEPSY_SOURCE}" parakeet)
add_library(hearth_transcribepsy SHARED "${HEARTH_CPP}/speech/backends/transcribepsy_backend.cpp")
target_compile_definitions(hearth_transcribepsy PRIVATE HEARTH_TRANSCRIBEPSY_ACCELERATION="cpu")
target_compile_options(hearth_transcribepsy PRIVATE "/FI${HEARTH_WINDOWS_COMPAT}/hearth_windows_compat.h")
target_link_libraries(hearth_transcribepsy PRIVATE qvac-parakeet)
target_link_options(hearth_transcribepsy PRIVATE /EXPORT:hearth_speech_backend_v1)
