# Windows x64 twin of scripts/translation/ct2/CMakeLists.txt. That recipe is part of the
# Android CT2 runtime receipt, so the MSVC differences live here instead of in it.
cmake_minimum_required(VERSION 3.22)
project(hearth_ct2_windows LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${BPE_LIB}")
 message(FATAL_ERROR "Build the pinned Rust tokenizer for x86_64-pc-windows-msvc first")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_MKL OFF CACHE BOOL "" FORCE)
set(WITH_DNNL OFF CACHE BOOL "" FORCE)
set(WITH_CUDA OFF CACHE BOOL "" FORCE)
set(WITH_CUDNN OFF CACHE BOOL "" FORCE)
set(WITH_HIP OFF CACHE BOOL "" FORCE)
set(WITH_RUY ON CACHE BOOL "" FORCE)
set(OPENMP_RUNTIME NONE CACHE STRING "" FORCE)
# x86-64 kernels are compiled for AVX/AVX2/AVX-512 and selected at runtime.
set(ENABLE_CPU_DISPATCH ON CACHE BOOL "" FORCE)
add_subdirectory("${CT2_SOURCE}" ct2)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${SPM_SOURCE}" spm EXCLUDE_FROM_ALL)
# The static Rust library names these Windows components (cargo rustc --print native-static-libs).
set(HEARTH_RUST_SYSTEM_LIBRARIES kernel32 ntdll userenv ws2_32 dbghelp synchronization bcrypt advapi32)
find_package(JNI REQUIRED)
add_library(hearth_desktop_ct2 SHARED "${CPP}/translation/ct2/ct2_jni.cpp")
target_compile_options(hearth_desktop_ct2 PRIVATE /utf-8)
target_include_directories(hearth_desktop_ct2 PRIVATE ${JNI_INCLUDE_DIRS} "${SPM_SOURCE}/src" "${CT2_SOURCE}/third_party")
target_link_libraries(hearth_desktop_ct2 PRIVATE ctranslate2 sentencepiece-static "${BPE_LIB}" ${HEARTH_RUST_SYSTEM_LIBRARIES})
add_executable(ct2_smoke "${CPP}/translation/ct2/ct2_smoke.cpp")
target_compile_options(ct2_smoke PRIVATE /utf-8)
target_include_directories(ct2_smoke PRIVATE "${SPM_SOURCE}/src" "${CT2_SOURCE}/third_party")
target_link_libraries(ct2_smoke PRIVATE ctranslate2 sentencepiece-static "${BPE_LIB}" ${HEARTH_RUST_SYSTEM_LIBRARIES})
