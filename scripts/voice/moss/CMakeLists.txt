cmake_minimum_required(VERSION 3.22)
project(hearth_moss_runtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${SPM_SOURCE}" spm EXCLUDE_FROM_ALL)
add_library(onnx_prebuilt SHARED IMPORTED)
set_target_properties(onnx_prebuilt PROPERTIES IMPORTED_LOCATION "${ORT_LIBRARY}")
add_library(hearth_moss SHARED "${MOSS_CPP}/moss.cpp")
target_include_directories(hearth_moss PRIVATE "${SPM_SOURCE}/src" "${ORT_HEADERS}")
target_link_libraries(hearth_moss PRIVATE sentencepiece-static onnx_prebuilt)
if(ANDROID)
    target_link_libraries(hearth_moss PRIVATE log)
    target_link_options(hearth_moss PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--version-script=${MOSS_CPP}/runtime.map" "-Wl,--exclude-libs,ALL")
endif()
if(NOT ANDROID)
    add_executable(moss_smoke "${CMAKE_CURRENT_SOURCE_DIR}/moss-smoke.cpp")
    target_include_directories(moss_smoke PRIVATE "${MOSS_CPP}/vendor")
    target_link_libraries(moss_smoke PRIVATE hearth_moss)
endif()
