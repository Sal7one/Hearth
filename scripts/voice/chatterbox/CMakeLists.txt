cmake_minimum_required(VERSION 3.22)
project(hearth_chatterbox_runtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
add_library(onnx_prebuilt SHARED IMPORTED)
set_target_properties(onnx_prebuilt PROPERTIES IMPORTED_LOCATION "${ORT_LIBRARY}")
add_library(hearth_bpe STATIC IMPORTED)
set_target_properties(hearth_bpe PROPERTIES IMPORTED_LOCATION "${BPE_LIBRARY}")
add_library(hearth_chatterbox SHARED "${CHATTERBOX_CPP}/chatterbox.cpp" "${UPSTREAM}/src/audio.cpp" "${WATERMARK_CPP}")
target_include_directories(hearth_chatterbox PRIVATE "${ORT_HEADERS}" "${UPSTREAM}/include")
target_link_libraries(hearth_chatterbox PRIVATE onnx_prebuilt hearth_bpe)
if(ANDROID)
 target_link_libraries(hearth_chatterbox PRIVATE log dl m)
 target_link_options(hearth_chatterbox PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--version-script=${CHATTERBOX_CPP}/runtime.map" "-Wl,--exclude-libs,ALL")
else()
 target_compile_definitions(hearth_chatterbox PRIVATE HEARTH_CHATTERBOX_TRACE=1)
 add_executable(chatterbox_smoke "${CMAKE_CURRENT_SOURCE_DIR}/chatterbox-smoke.cpp")
 target_include_directories(chatterbox_smoke PRIVATE "${CHATTERBOX_CPP}/vendor")
 target_link_libraries(chatterbox_smoke PRIVATE hearth_chatterbox)
endif()
