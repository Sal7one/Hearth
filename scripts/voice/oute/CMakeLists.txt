cmake_minimum_required(VERSION 3.22)
project(hearth_oute LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
add_subdirectory("${OUTE_SOURCE}/ggml" ggml)
set(core backend chat vectorstore layers tokenizer models unicode unicode-data vision_process audio_process)
set(sources)
foreach(name IN LISTS core)
 list(APPEND sources "${OUTE_SOURCE}/src/${name}.cpp")
endforeach()
foreach(name llama qwen qwen_asr orpheus oute)
 list(APPEND sources "${OUTE_SOURCE}/models/${name}.cpp")
endforeach()
add_library(hearth_oute SHARED ${sources} "${HEARTH_ROOT}/scripts/voice/oute/runtime.cpp")
target_include_directories(hearth_oute PRIVATE "${HEARTH_ROOT}/scripts/voice/oute" "${OUTE_SOURCE}/src" "${OUTE_SOURCE}/models" "${HEARTH_ROOT}/hearth-tts-oute/src/main/cpp/vendor")
target_link_libraries(hearth_oute PRIVATE ggml)
target_compile_options(hearth_oute PRIVATE -fvisibility=hidden -O2)
if(APPLE)
 target_link_options(hearth_oute PRIVATE -Wl,-exported_symbol,_hearth_oute_abi -Wl,-exported_symbol,_hearth_oute_last_error -Wl,-exported_symbol,_hearth_oute_create -Wl,-exported_symbol,_hearth_oute_synthesize -Wl,-exported_symbol,_hearth_oute_cancel -Wl,-exported_symbol,_hearth_oute_destroy)
else()
 target_link_options(hearth_oute PRIVATE "-Wl,--version-script=${HEARTH_ROOT}/scripts/voice/oute/exports.map" -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,-z,defs)
endif()
set_target_properties(ggml ggml-base ggml-cpu PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
