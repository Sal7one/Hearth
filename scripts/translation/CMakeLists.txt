cmake_minimum_required(VERSION 3.22)
project(transiber_translation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(ANDROID)
 set(GGML_METAL OFF CACHE BOOL "" FORCE)
else()
 option(HEARTH_HOST_METAL "Enable Apple Metal for local Mac benchmarks" OFF)
 set(GGML_METAL ${HEARTH_HOST_METAL} CACHE BOOL "" FORCE)
endif()
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
if(ANDROID)
 add_library(transiber_translation SHARED "${CPP}/translation/translation_jni.cpp")
 target_link_libraries(transiber_translation PRIVATE llama log)
 target_link_options(transiber_translation PRIVATE -Wl,--exclude-libs,ALL "-Wl,--version-script=${CPP}/translation/exports.map" -Wl,-z,max-page-size=16384)
else()
 add_executable(translation_smoke "${CPP}/translation/translation_smoke.cpp")
 target_link_libraries(translation_smoke PRIVATE llama)
endif()
