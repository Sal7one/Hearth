cmake_minimum_required(VERSION 3.22)
project(transiber_translation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(ANDROID)
 set(GGML_METAL OFF CACHE BOOL "" FORCE)
 option(HEARTH_ANDROID_VULKAN "Build the separately loaded ARM64 Vulkan runtime" OFF)
 option(HEARTH_ANDROID_OPENCL "Build the separately loaded ARM64 Adreno OpenCL runtime" OFF)
 set(GGML_VULKAN ${HEARTH_ANDROID_VULKAN} CACHE BOOL "" FORCE)
 set(GGML_OPENCL ${HEARTH_ANDROID_OPENCL} CACHE BOOL "" FORCE)
 if((HEARTH_ANDROID_VULKAN OR HEARTH_ANDROID_OPENCL) AND NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "The optional GPU runtimes are packaged for ARM64 only")
 endif()
 if(HEARTH_ANDROID_VULKAN AND HEARTH_ANDROID_OPENCL)
  message(FATAL_ERROR "Each private runtime must contain exactly one GPU backend")
 endif()
 if(HEARTH_ANDROID_OPENCL)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(OPENCL_SOURCE "${LLAMA_SOURCE}/ggml/src/ggml-opencl")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/generate-opencl-shim.py"
   --headers "${HEARTH_OPENCL_HEADERS}" --source "${OPENCL_SOURCE}/ggml-opencl.cpp"
   --source "${OPENCL_SOURCE}/cl-program-cache.cpp" --source "${CPP}/common/opencl_loader.h"
   --output "${CMAKE_CURRENT_BINARY_DIR}/opencl-shim.cpp" COMMAND_ERROR_IS_FATAL ANY)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/patch-opencl-backend.py"
   --source "${OPENCL_SOURCE}/ggml-opencl.cpp" --output "${CMAKE_CURRENT_BINARY_DIR}/opencl-patched.cpp"
   COMMAND_ERROR_IS_FATAL ANY)
  add_library(hearth_opencl_driver STATIC "${CMAKE_CURRENT_BINARY_DIR}/opencl-shim.cpp")
  target_include_directories(hearth_opencl_driver PRIVATE "${HEARTH_OPENCL_HEADERS}" "${CPP}/common")
  target_link_libraries(hearth_opencl_driver PRIVATE dl)
  # FindOpenCL's library variable can name a CMake target. Our typed shim has
  # no DT_NEEDED entry for the device-vendor libOpenCL.so.
  set(OpenCL_LIBRARY hearth_opencl_driver CACHE STRING "" FORCE)
  set(OpenCL_INCLUDE_DIR "${HEARTH_OPENCL_HEADERS}" CACHE PATH "" FORCE)
  set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
  set(GGML_OPENCL_USE_ADRENO_KERNELS ON CACHE BOOL "" FORCE)
  set(GGML_OPENCL_USE_ADRENO_BIN_KERNELS OFF CACHE BOOL "" FORCE)
  set(GGML_OPENCL_PROFILING OFF CACHE BOOL "" FORCE)
 endif()
else()
 option(HEARTH_HOST_METAL "Enable Apple Metal for local Mac benchmarks" OFF)
 option(HEARTH_HOST_VULKAN "Enable Vulkan for host validation only" OFF)
 set(GGML_METAL ${HEARTH_HOST_METAL} CACHE BOOL "" FORCE)
 set(GGML_VULKAN ${HEARTH_HOST_VULKAN} CACHE BOOL "" FORCE)
 set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
endif()
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(GGML_VULKAN)
 find_package(SPIRV-Headers CONFIG REQUIRED)
endif()
add_subdirectory("${LLAMA_SOURCE}" llama)
if(GGML_OPENCL)
 get_target_property(OPENCL_SOURCES ggml-opencl SOURCES)
 list(REMOVE_ITEM OPENCL_SOURCES ggml-opencl.cpp)
 set_property(TARGET ggml-opencl PROPERTY SOURCES "${OPENCL_SOURCES}")
 target_sources(ggml-opencl PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/opencl-patched.cpp")
 target_include_directories(ggml-opencl PRIVATE "${OPENCL_SOURCE}")
endif()
if(GGML_VULKAN)
 target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif()
if(ANDROID)
 set(TRANSLATION_TARGET transiber_translation)
 set(TRANSLATION_EXPORTS exports.map)
 if(HEARTH_ANDROID_VULKAN)
  set(TRANSLATION_TARGET transiber_translation_vulkan)
  set(TRANSLATION_EXPORTS vulkan_exports.map)
 endif()
 if(HEARTH_ANDROID_OPENCL)
  set(TRANSLATION_TARGET transiber_translation_opencl)
  set(TRANSLATION_EXPORTS opencl_exports.map)
 endif()
 add_library(${TRANSLATION_TARGET} SHARED "${CPP}/translation/translation_jni.cpp")
 if(HEARTH_ANDROID_VULKAN)
  target_compile_definitions(${TRANSLATION_TARGET} PRIVATE HEARTH_TRANSLATION_VULKAN=1)
 endif()
 if(HEARTH_ANDROID_OPENCL)
  target_compile_definitions(${TRANSLATION_TARGET} PRIVATE HEARTH_TRANSLATION_OPENCL=1)
  target_include_directories(${TRANSLATION_TARGET} PRIVATE "${HEARTH_OPENCL_HEADERS}")
 endif()
 target_link_libraries(${TRANSLATION_TARGET} PRIVATE llama log)
 target_link_options(${TRANSLATION_TARGET} PRIVATE -Wl,--exclude-libs,ALL "-Wl,--version-script=${CPP}/translation/${TRANSLATION_EXPORTS}" -Wl,-z,max-page-size=16384)
else()
 add_executable(translation_smoke "${CPP}/translation/translation_smoke.cpp")
 target_link_libraries(translation_smoke PRIVATE llama)
 # Weight-free regression tests compile the production wrapper against the
 # exact runtime headers, replacing only C API entry points with test doubles.
 add_executable(translation_runtime_test "${CPP}/translation/tests/text_model_test.cpp")
 target_include_directories(translation_runtime_test PRIVATE
  "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include")
 target_compile_options(translation_runtime_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 enable_testing()
 add_test(NAME translation_runtime_test COMMAND translation_runtime_test)
 add_executable(translation_vulkan_policy_test "${CPP}/translation/tests/vulkan_policy_test.cpp")
 target_compile_options(translation_vulkan_policy_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 add_test(NAME translation_vulkan_policy_test COMMAND translation_vulkan_policy_test)
 add_executable(translation_opencl_policy_test "${CPP}/translation/tests/opencl_policy_test.cpp")
 target_compile_options(translation_opencl_policy_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 add_test(NAME translation_opencl_policy_test COMMAND translation_opencl_policy_test)
endif()
