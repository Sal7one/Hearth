cmake_minimum_required(VERSION 3.22)
project(hearth_ct2 LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_MKL OFF CACHE BOOL "" FORCE)
set(WITH_DNNL OFF CACHE BOOL "" FORCE)
set(WITH_CUDA OFF CACHE BOOL "" FORCE)
set(WITH_CUDNN OFF CACHE BOOL "" FORCE)
set(WITH_HIP OFF CACHE BOOL "" FORCE)
set(WITH_RUY ON CACHE BOOL "" FORCE)
set(OPENMP_RUNTIME NONE CACHE STRING "" FORCE)
set(ENABLE_CPU_DISPATCH OFF CACHE BOOL "" FORCE)
add_subdirectory("${CT2_SOURCE}" ct2)
set_target_properties(ctranslate2 PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/patch-thread-pool.py" "${CT2_SOURCE}/src/thread_pool.cc" "${CMAKE_CURRENT_BINARY_DIR}/thread_pool_android.cc" COMMAND_ERROR_IS_FATAL ANY)
 get_target_property(CT2_SOURCES ctranslate2 SOURCES)
 list(REMOVE_ITEM CT2_SOURCES src/thread_pool.cc)
 set_property(TARGET ctranslate2 PROPERTY SOURCES "${CT2_SOURCES}")
 target_sources(ctranslate2 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/thread_pool_android.cc")
endif()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${SPM_SOURCE}" spm EXCLUDE_FROM_ALL)
if(ANDROID)
 add_library(transiber_ct2 SHARED "${CPP}/translation/ct2/ct2_jni.cpp")
 target_include_directories(transiber_ct2 PRIVATE "${SPM_SOURCE}/src")
 target_link_libraries(transiber_ct2 PRIVATE ctranslate2 sentencepiece-static log)
 target_link_options(transiber_ct2 PRIVATE -Wl,--exclude-libs,ALL "-Wl,--version-script=${CPP}/translation/ct2/exports.map" -Wl,-z,max-page-size=16384)
else()
 add_executable(ct2_smoke "${CPP}/translation/ct2/ct2_smoke.cpp")
 target_include_directories(ct2_smoke PRIVATE "${SPM_SOURCE}/src")
 target_link_libraries(ct2_smoke PRIVATE ctranslate2 sentencepiece-static)
endif()
