cmake_minimum_required(VERSION 3.22.1)
project(hearth_supertonic_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../..")
add_library(onnx_prebuilt SHARED IMPORTED)
set_target_properties(onnx_prebuilt PROPERTIES IMPORTED_LOCATION "${REPOSITORY_ROOT}/hearth-onnx-runtime/src/main/jniLibs/${ANDROID_ABI}/libonnxruntime.so")
add_library(hearth_supertonic_jni SHARED voice/voice_jni.cpp voice/supertonic.cpp)
target_include_directories(hearth_supertonic_jni PRIVATE "${REPOSITORY_ROOT}/common-jni/src/main/cpp/common" "${REPOSITORY_ROOT}/common-jni/src/main/cpp/onnx/include")
target_link_libraries(hearth_supertonic_jni PRIVATE onnx_prebuilt)
target_compile_options(hearth_supertonic_jni PRIVATE -fvisibility=hidden -fstack-protector-strong)
target_link_options(hearth_supertonic_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
